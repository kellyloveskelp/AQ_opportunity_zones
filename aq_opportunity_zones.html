Oyster Reclassification, Masking, and Mapping Function

aq_opportunity_zones <- function(species, sst_min, sst_max, depth_min, depth_max){

    # SST Reclassification Matrix

    sst_rcl <- matrix(c(-Inf, sst_min, 0, # values -Inf to min = 0
                        sst_min, sst_max, 1,  # values min to max = 1
                        sst_max, Inf, 0), # values max to Inf = 0
                        ncol = 3, byrow = TRUE)

    # Depth Reclassification Matrix

    depth_rcl <- matrix(c(-Inf, depth_min, 0, # values -Inf to min = 0
                        depth_min, depth_max, 1,  # values min to max = 1
                        depth_max, Inf, 0), # values max to Inf = 0
                        ncol = 3, byrow = TRUE)

    # Apply reclasification matrices
    optimal_sst <- terra::classify(avg_sst, rcl = sst_rcl)
    optimal_depth <- terra::classify(depth_crop_resamp, rcl = depth_rcl)

    # Find locations where both SST AND depth AND EEZ = 1
    optimal_cond <- optimal_sst * optimal_depth

    # Rasterize EEZ
    eez_raster <- rasterize(eez, optimal_cond, field = 'rgn_key')

    # Mask optimal conditions by EEZ
    optimal_zones <- mask(optimal_cond, eez_raster)
                         
    area_by_eez <- (zonal(optimal_zones, eez_raster, fun = 'sum', na.rm = TRUE)) |> 
        as_tibble() |> 
        as.data.frame() |> 
        rename(sum = mean)

    # Join area dataset to eez dataset to use geometries, apply st_as_sf() to plot
    opportunity_zones <- inner_join(area_by_eez, eez, by = 'rgn_key') |> 
        st_as_sf() |> 
        select(rgn, area_km2)

    # Display oppotunity_zones in easy-to-read table
    kable_table <- kable(
        opportunity_zones, 
        format = 'simple', 
        col.names = c("Region", "Area (km^2)", "Geometry"),
        caption = paste("Oppotunity Area by Region for", species, "Aquaculture")
        )

    ## Plot !
    opportunity_map <- tm_basemap("Esri.OceanBasemap") +
        tm_shape(opportunity_zones) +
        tm_options(component.autoscale = FALSE) +
        # Map exclusive economic zones by suitable area                     
        tm_polygons(fill = 'area_km2', lwd = 0.7, fill_alpha = 0.6,
                fill.scale = tm_scale_categorical(values = 'brewer.yl_or_rd'),
                fill.legend = tm_legend(title = "Optimal Area (km^2)",
                                        position = tm_pos_out('left'))) +
        tm_compass(position = c('left', 'bottom'), # Add compass
                    size = 2) +
        tm_scalebar(position = c('left', 'bottom')) + # Add scalebar
        tm_title(text = paste("Opportunity Zones for", species, "Aquaculture along the US West Coast"),
                size = 1,
                position = tm_pos_out('center', 'top')) +
        tm_labels(text = 'rgn', size = 0.5, fontface = 'bold', position = tm_pos_out('center'))

    print(kable_table)
    opportunity_map
}
